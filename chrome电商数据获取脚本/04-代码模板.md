# 04 · 代码模板

最小可用模板：读 Cookie → GET HTML 解析 token → 直接调接口。全程不开 tab、不渲染 DOM。

依赖：

```bash
pip install websockets
```

## 完整示例

```python
# -*- coding: utf-8 -*-
"""无 tab 获取生意参谋数据的最小模板。"""
import asyncio
import json
import re
import time
import urllib.parse
import urllib.request

import websockets

PORT = 9223
BASE = "https://sycm.taobao.com"
ITEM_ID = "731918476865"
DATE_RANGE = "2026-11-01|2026-11-01"


# ---------- 1. Cookie ----------
def get_browser_info() -> dict:
    with urllib.request.urlopen(f"http://localhost:{PORT}/json/version", timeout=10) as r:
        return json.loads(r.read().decode())


async def read_cookies() -> list[dict]:
    info = get_browser_info()
    async with websockets.connect(info["webSocketDebuggerUrl"], open_timeout=10) as ws:
        await ws.send(json.dumps({"id": 1, "method": "Storage.getCookies"}))
        while True:
            msg = json.loads(await asyncio.wait_for(ws.recv(), 10))
            if msg.get("id") == 1:
                return msg["result"]["cookies"]


def build_cookie_header(cookies: list[dict]) -> str:
    now = time.time()
    seen = {}
    for c in cookies:
        domain = c.get("domain", "")
        if "taobao.com" not in domain and "tmall.com" not in domain:
            continue
        expires = c.get("expires", -1)
        if isinstance(expires, (int, float)) and 0 < expires < now:
            continue
        name, value = c.get("name"), c.get("value")
        if not name:
            continue
        score = 2 if domain == "sycm.taobao.com" else 1
        if name not in seen or score > seen[name][0]:
            seen[name] = (score, value)
    return "; ".join(f"{k}={v}" for k, (_, v) in seen.items())


# ---------- 2. 请求头 ----------
def make_headers(cookie: str, referer: str) -> dict:
    info = get_browser_info()
    return {
        "Cookie": cookie,
        "User-Agent": info.get("User-Agent", "Mozilla/5.0"),
        "Referer": referer,
        "Accept": "application/json, text/plain, */*",
        "Origin": BASE,
        "x-requested-with": "XMLHttpRequest",
        "Sec-Fetch-Site": "same-origin",
        "Sec-Fetch-Mode": "cors",
        "Sec-Fetch-Dest": "empty",
    }


# ---------- 3. HTML + token ----------
def get_html(url: str, headers: dict) -> str:
    req = urllib.request.Request(url, headers={**headers, "Accept": "text/html"})
    with urllib.request.urlopen(req, timeout=20) as r:
        return r.read().decode("utf-8", "replace")


def parse_token(html: str) -> str:
    m = re.search(r'<meta name="microdata" content="([^"]*)"', html)
    if not m:
        raise RuntimeError("找不到 meta[microdata]，可能被重定向到登录页")
    mm = re.search(r"(?:^|;)legalityToken=([^;]*)", m.group(1))
    if not mm:
        raise RuntimeError("microdata 里没有 legalityToken")
    return mm.group(1)


# ---------- 4. 调接口 ----------
def api_url(path: str, params: dict) -> str:
    return BASE + path + "?" + urllib.parse.urlencode(params)


def get_json(url: str, headers: dict) -> dict:
    req = urllib.request.Request(url, headers=headers)
    with urllib.request.urlopen(req, timeout=20) as r:
        return json.loads(r.read().decode("utf-8", "replace"))


def check(obj: dict) -> dict:
    code = obj.get("code")
    if code != 0:
        raise RuntimeError(f"接口返回 code={code}, message={obj.get('message')}")
    return obj


# ---------- 5. 示例：flow tab ----------
async def demo_flow():
    cookies = await read_cookies()
    headers = make_headers(
        build_cookie_header(cookies),
        f"{BASE}/cc/item_archives?activeKey=flow&itemId={ITEM_ID}",
    )

    page_url = (
        f"{BASE}/cc/item_archives?activeKey=flow&itemId={ITEM_ID}"
        f"&dateType=day&dateRange={urllib.parse.quote(DATE_RANGE)}"
    )
    token = parse_token(get_html(page_url, headers))
    ts = int(time.time() * 1000)

    base_params = {
        "itemId": ITEM_ID,
        "dateType": "day",
        "dateRange": DATE_RANGE,
        "_": ts,
        "token": token,
    }

    # Top10 流量来源
    params = {**base_params, "page": 1, "pageSize": 10}
    data = check(get_json(api_url("/cc/item/flow/top/v4.json", params), headers))
    print("flow top rows:", len(data.get("data", [])))
    print(json.dumps(data["data"][:2], ensure_ascii=False, indent=2))

    time.sleep(1)  # 控制频率

    # Top10 搜索词
    params = {**base_params, "page": 1, "pageSize": 10}
    data = check(get_json(api_url("/cc/item/searchword/top/v3.json", params), headers))
    print("searchword rows:", len(data.get("data", [])))
    print(json.dumps(data["data"][:2], ensure_ascii=False, indent=2))


if __name__ == "__main__":
    asyncio.run(demo_flow())
```

async def demo_sale():

    headers = make_headers(
        build_cookie_header(cookies),

    page_url = (
        f"{BASE}/cc/item_archives?activeKey=sale&itemId={ITEM_ID}"

    )
    token = parse_token(get_html(page_url, headers))
    ts = int(time.time() * 1000)

    params = {
        "itemId": ITEM_ID,
        "dateType": "day",
        "dateRange": DATE_RANGE,
        "device": 0,
        "_": ts,
        "token": token,
    }
    data = check(get_json(api_url("/cc/item/sale/overview.json", params), headers))
    print(json.dumps(data["data"], ensure_ascii=False, indent=2))

## 注意事项

1. **不要硬编码 token**
   - 每次跑都从 HTML 的 `meta[microdata]` 解析。
   - token 失效时重新 GET 一次 HTML。

2. **Cookie 从浏览器读**
   - `Storage.getCookies` 是浏览器级命令，不创建 tab。
   - 也可以导出到本地 cookie jar，但要处理过期。

3. **控制频率**
   - 串行请求，每个请求之间 `sleep(1~3)` 秒。
   - 不要并发几十个请求。

4. **处理 1003**
   - `code=1003` 表示缺参数，按 `message` 补：
     - `device not exist` → 加 `device=2`
     - `belong not exist` → 加 `belong=all`
     - `order not exist` → 加 `order=desc&orderBy=uv`
     - `pageId not exist` → 加 `pageId`/`pPageId`

5. **处理登录失效**
   - 返回 HTML 登录页或 HTTP 401/403：
     - 重新登录 Chrome
     - 重新读 Cookie
     - 重新 GET HTML 取 token

6. **保存接口模板**
   - 把验证过的接口和参数写进 `endpoints.json`。
   - 下次直接读配置，不需要重新找 bundle。
